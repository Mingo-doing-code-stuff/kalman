% KALMAN FILTER FIRST EXPLAINATION
\begin{frame}
    \frametitle{What is the Kalman Filter?}
    The Kalman filter is a mathematical method for recursively estimating parameters to describe
    system states.

    In this process, a prediction about a parameter value is repeatedly made, combined with the
    error-prone measurement, and then used again to make a new prediction.
\end{frame}

% THE COVARIANCE
\begin{frame}
    \frametitle{Covariance}
    \footnotesize

    \begin{itemize}
        \item \textbf{Definition:}
              Covariance is a measure of how much two random variables change together. If the greater values of one variable mainly correspond with the greater values of the other variable, and the same holds for the lesser values (i.e., the variables tend to show similar behavior), the covariance is positive. In the opposite case, when the greater values of one variable mainly correspond to the lesser values of the other (i.e., the variables tend to show opposite behavior), the covariance is negative.
        \item \textbf{Mathematical Definition}
              Given two random variables \(X\) and \(Y\), the covariance is defined as:\\
              \[
                  \text{Cov}(X, Y) = \frac{\sum_{i=1}^{N} (x_{i} - \overline{x})(y_{i} - \overline{y})}{N - 1}
              \] \\
              where \(\overline{x}\) and \(\overline{y}\) are the means of \(X\) and \(Y\), respectively.
    \end{itemize}

\end{frame}

% THE COVARIANCE, TOO
\begin{frame}
    \frametitle{The covariance}
    \begin{itemize}
        \item \textbf{Example:}
            Consider two random variables, \(X\) and \(Y\), representing the number of hours studied and the test scores of students. If higher hours of study tend to correspond with higher test scores, the covariance between \(X\) and \(Y\) would be positive.
        \item \textbf{Properties}
              \begin{itemize}
                  \item \(\text{Cov}(X, Y) = \text{Cov}(Y, X)\)
                  \item \(\text{Cov}(X, X) = \text{Var}(X)\) (Variance of \(X\))
                  \item If \(X\) and \(Y\) are independent, \(\text{Cov}(X, Y) = 0\)
              \end{itemize}
    \end{itemize}


\end{frame}

\section{The Math behind}

\begin{frame}
    \frametitle{State Representation}
    % //TODO: Make x's differentiabable
    We define the state vector \(x\) to include the position and velocity in both x and y directions.

    \begin{equation*}
        x = \begin{bmatrix}
            x     \\
            y     \\
            v_{x} \\
            v_{y} \\
        \end{bmatrix}
    \end{equation*}

    where:
    \begin{itemize}
        \item \(x\) is the position on the x-axis
        \item \(y\) is the postion on the y-axis
        \item \(v_{x}\) is the velocity in the x-direction
        \item \(v_{y}\) is the velocity in the y-direction

    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Formulate the Transtition Model}
    The state transition model describes how the state evolves from one time step to the next.
    If we assume a constant velocity model, the state transition can be expressed as:
    \begin{equation*}
        x_{k+1}=Ax_{k}+w_{k}
    \end{equation*}
    where \(w_{k}\) represents the process noise, assumed to be Gaussian with zero mean and covariance \(Q\).
\end{frame}

\begin{frame}
    \frametitle{Create the Transition Matrix}
    The transition matrix A for a constant velocity Model is:

    \begin{equation*}
        A = \begin{bmatrix}
            1 & 0 & \Delta t & 0        \\
            0 & 1 & 0        & \Delta t \\
            0 & 0 & 1        & 0        \\
            0 & 0 & 0        & 1        \\
        \end{bmatrix}
    \end{equation*}

    \(\Delta t\) describes the changes between two measurements.
\end{frame}

\begin{frame}
    \frametitle{Observation Model}
    The observation model realtes the state to the measurements:
    \begin{equation*}
        z_{k}=Hx_{k}+v_{k}
    \end{equation*}
    where \(v_{k}\) represents the measurement noise, assumed to be Gaussian with zero mean and
    covariance \(R\).

    The observation matrix \(H\) for direct measurement of position is:

    \begin{equation*}
        H = \begin{bmatrix}
            1 & 0 & 0 & 0 \\
            0 & 1 & 0 & 0 \\
        \end{bmatrix}
    \end{equation*}

\end{frame}

\begin{frame}
    \frametitle{Prediction Step}

    The prediction step estimates the next state and its uncertainty:

    \begin{itemize}
        \item Predict the state:
              \begin{equation}
                  x_{k+1}=Ax_{k} + Bu_{k}
              \end{equation}

        \item Predict the error coavriance:
              \begin{equation}
                  P_{k+1}=AP_{k}A^{T}+Q
              \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Correction Step}
    The correction step updates the state estimate with the new measurement:
    \begin{itemize}
        \item Compute the Kalman gain:
              \begin{equation}
                  K_{k} = P_{k}H^{T}(HP_{k}H^{T}+R)^{-1}
              \end{equation}
        \item Update the state estimate:
              \begin{equation}
                  x_{k} = x_{k}+K_{k}(z_{k}-Hx_{k})
              \end{equation}
        \item Update the error covariance:
              \begin{equation}
                  P_{k} = (I-K_{k}H)P_{k}
              \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Explaination to simplified Kalman filter}

    \begin{itemize}
        \item Prediction:
              \begin{itemize}
                  \item{\(A\): Resolves to \(1\) due to the assumption of constant velocity}
                  \item{\(Bu_{k}\): Resolves to \(0\) due to no external Control-Signals}
                  \item{\(Q\): Resolves to \(0\) due to the assumption of absence of noise caused by external factors}
              \end{itemize}
        \item Correction:
              \begin{itemize}
                  \item{\(H\): Resolves to \(1\) due to the assumption that a measurement consists of a state value and noise}
                  \item{\(I\): Resolves to \(1\) due to beeing a unit matrix}
              \end{itemize}
    \end{itemize}

\end{frame}

\begin{frame}
    \frametitle{Prediction Step}

    The prediction step estimates the next state and its uncertainty:

    \begin{itemize}
        \item Predict the state: \\
              \(x_{k+1}=Ax_{k} + Bu_{k}=1 \cdot x_{k} + 0 \cdot u_{k} = \bf{x_{k}}\)
        \item Predict the error covariance: \\
              \(P_{k+1}=AP_{k}A^{T}+Q=1 \cdot P_{k} \cdot 1 + 0 = \bf{P_{k}}\)
    \end{itemize}

    The correction step updates the state estimate with the new measurement:
    \begin{itemize}
        \item Compute the Kalman gain: \\
              \(K_{k} = P_{k}H^{T}(HP_{k}H^{T}+R)^{-1} = \frac{P_{k} \cdot 1} {1 \cdot P_{k} \cdot 1 + R} = \bf{\frac{P_{k}} {P_{k} + R}}\)
        \item Update the state estimate: \\
              \(x_{k} = x_{k} + K_{k}(z_{k} - Hx_{k}) = x_{k} + K_{k} \cdot (z_{k} - 1 \cdot x_{k}) = \bf{x_{k} + K_{k} \cdot (z_{k} - x_{k})}\)
        \item Update the error covariance: \\
              \(P_{k} = (I - K_{k}H)P_{k} = (1 - K_{k} \cdot 1) \cdot P_{k} = \bf{(1 - K_{k}) \cdot P_{k}}\)
    \end{itemize}
\end{frame}

\begin{frame}{Types of Kalman Filters}
    \footnotesize
    \begin{itemize}
        \item \textbf{Kalman Filter (KF)}
              \begin{itemize}
                  \item Assumes linear system dynamics and Gaussian noise
                  \item Optimal for linear systems
              \end{itemize}
        \item \textbf{Extended Kalman Filter (EKF)}
              \begin{itemize}
                  \item Handles non-linear system dynamics by linearizing around the current estimate
                  \item Uses first-order Taylor expansion
              \end{itemize}
        \item \textbf{Unscented Kalman Filter (UKF)}
              \begin{itemize}
                  \item Handles non-linear system dynamics without linearization
                  \item Uses a deterministic sampling approach to capture the mean and covariance accurately
              \end{itemize}
        \item \textbf{Ensemble Kalman Filter (EnKF)}
              \begin{itemize}
                  \item Uses an ensemble of system states to estimate the mean and covariance
                  \item Suitable for high-dimensional systems
              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Differences between Kalman Filter Types}
    \tiny
    \begin{itemize}
        \item \textbf{Linear vs Non-linear}
              \begin{itemize}
                  \item KF: Linear systems
                  \item EKF: Non-linear systems, linearized
                  \item UKF: Non-linear systems, unscented transformation
                  \item EnKF: Non-linear systems, ensemble approach
              \end{itemize}
        \item \textbf{Approach to Non-linearity}
              \begin{itemize}
                  \item EKF: Linearization using Taylor series
                  \item UKF: Deterministic sampling (unscented transform)
                  \item EnKF: Monte Carlo sampling (ensemble)
              \end{itemize}
        \item \textbf{Computational Complexity}
              \begin{itemize}
                  \item KF: Low
                  \item EKF: Moderate, due to linearization
                  \item UKF: Moderate, due to unscented transform
                  \item EnKF: High, due to multiple ensemble members
              \end{itemize}
        \item \textbf{Accuracy in Non-linear Systems}
              \begin{itemize}
                  \item EKF: May be less accurate due to linearization
                  \item UKF: More accurate, captures non-linearities better
                  \item EnKF: High accuracy, handles complex non-linearities well
              \end{itemize}
    \end{itemize}
\end{frame}
